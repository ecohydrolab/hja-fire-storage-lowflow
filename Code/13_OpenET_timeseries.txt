// Inputs
// ================================================================
var HFF_polygons = ee.FeatureCollection('users/duffshan/HFF_vegmort_polygons_Clip');
var LF_polygons = ee.FeatureCollection('users/duffshan/LF_vegmort_polygons_Clip');
var unburned_polygon = ee.FeatureCollection('users/duffshan/HJA_unburned');
var watersheds = ee.FeatureCollection('users/duffshan/Gauged_Watersheds');

var et = ee.ImageCollection(
    'projects/openet/assets/ensemble/conus/gridmet/monthly/v2_1')
  .filter(ee.Filter.eq('mgrs_tile', '10T'))
  .filterDate('2020-01-01', '2026-01-01')
  .select('et_ensemble_mad');

print('Images available:', et.size());


// ================================================================
// Calculate mean across watershed for every available month
// ================================================================

var results_watersheds = et.map(function(img) {

  var date = ee.Date(img.get('system:time_start'));
  var label = date.format('YYYY_MM');

  var stats_watersheds = img.reduceRegions({
    collection: watersheds,
    reducer: ee.Reducer.mean().combine({
      reducer2: ee.Reducer.stdDev(),
      sharedInputs: true
    })
    .combine({
      reducer2: ee.Reducer.count(),
      sharedInputs: true
    }),
    scale: 30
  });

  return stats_watersheds.map(function(f) {
    var sd=ee.Number(f.get('stdDev'));
    var n = ee.Number(f.get('count'));
    var se = sd.divide(n.sqrt());
    return f.set({
      'month': label,
      'date': date.format('YYYY-MM-dd'),
      'standard_error': se
    });
  });

}).flatten();
